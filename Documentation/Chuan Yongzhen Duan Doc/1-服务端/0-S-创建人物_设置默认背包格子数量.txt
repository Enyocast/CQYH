// 初始化流程


当创建人物之后, 第一次进入游戏里, 走这个函数 加载数据
BOOL CRole::Load(const char* pcData, UINT32 dwSize)
{
	// 在这里判断人物等级, 初始化新角色 _InitNewRole
	if (GetLevel() == 0)
	{
		nRetCode = _InitNewRole();
		LOG_PROCESS_ERROR(nRetCode);
	}
}

BOOL CRole::_InitNewRole(void)
{
	// 初始化背包, 走这里
	m_ItemBag.Clear();
}

void CItemBag::Clear()
{
	// 在这里 对角色的所有容器做初始化
	for (INT32 i = 0; i < CONSLOT_COUNT; i++)
	{
		m_con_sizes[i] = INIT_CONTAINER_SIZE[i];
	}
}
